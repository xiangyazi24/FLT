ClearAll[b, c, u, q2sq, q7, h49, cross, half,
  pts, hull, edges, cr, onseg, faceData];

q2sq = b^4;
u[0] = 0;
u[1] = 1;
u[2] = 1;
u[3] = -b^3;
u[4] = -b^4*c;

u[n_Integer?Positive] /; n >= 5 := u[n] = Module[{k, m},
  If[EvenQ[n],
    k = n/2;
    Expand[
      u[k - 1]^2*u[k]*u[k + 2]
        - u[k - 2]*u[k]*u[k + 1]^2
    ],
    k = (n - 1)/2;
    m = k - 2;
    Expand[
      u[k + 2]*u[k]^3*If[EvenQ[m], q2sq, 1]
        - u[k - 1]*u[k + 1]^3*If[EvenQ[m], 1, q2sq]
    ]
  ]
];

q7 = c^3 + b*c - b^2;
h49 = Expand[Cancel[u[49]/(b^800*q7)]];
cr = CoefficientRules[h49, {b, c}];
pts = Sort[First /@ cr];

cross[o_, a_, z_] := Det[{a - o, z - o}];
half[list_] := Fold[
  Function[{stack, p}, Module[{s = stack},
    While[Length[s] >= 2 && cross[s[[-2]], s[[-1]], p] <= 0,
      s = Most[s]
    ];
    Append[s, p]
  ]],
  {}, list
];

hull = Join[Most[half[pts]], Most[half[Reverse[pts]]]];
edges = Partition[Append[hull, First[hull]], 2, 1];

onseg[p_, q_, r_] :=
  cross[p, q, r] == 0 &&
    And @@ MapThread[
      (Min[#1, #2] <= #3 <= Max[#1, #2]) &,
      {p, q, r}
    ];

faceData = Map[
  Function[e, Module[
    {p = e[[1]], q = e[[2]], d, normal, g,
     selected, face, initial},
    d = q - p;
    normal = {-d[[2]], d[[1]]};
    g = GCD @@ Abs[normal];
    normal = normal/g;
    selected = Select[cr, onseg[p, q, First[#]] &];
    face = Expand@Total[
      Last[#]*b^(First[#][[1]])*c^(First[#][[2]]) & /@ selected
    ];
    initial = PolynomialMod[face, 2];
    {
      p, q, normal,
      Factor[initial, Modulus -> 2],
      Mod[initial /. {b -> 1, c -> 1}, 2]
    }
  ]],
  edges
];
