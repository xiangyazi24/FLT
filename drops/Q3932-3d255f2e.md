# Q3932 — closing `CyclicExclusion20.lean`

Paste the following block after the already-proved helper lemmas in `CyclicExclusion20.lean`.  The first theorem is the kernel calculation for the coproduct map `ZMod 2 × ZMod 10 →+ H`; the second theorem is the final exclusion of a rational point of order `20`.

The proof follows the requested route: build the two `ZMod.lift` maps, use `AddMonoidHom.coprod`, reduce injectivity to kernel triviality, split the `ZMod 2` coordinate, and in the nonzero branch use `eq_five_nsmul_of_order_two_in_zmultiples` to contradict `eta_ne_half_image_20`.

```lean
import Mathlib

noncomputable section

open Function

private theorem zmod_two_ten_coprod_injective
    {H : Type*} [AddCommGroup H]
    {η Q : H}
    (hη : addOrderOf η = 2)
    (hQ : addOrderOf Q = 10)
    (hindep : η ≠ 5 • Q)
    (gη : ZMod 2 →+ H)
    (gQ : ZMod 10 →+ H)
    (hgη : ∀ z : ℤ, gη (z : ZMod 2) = z • η)
    (hgQ : ∀ z : ℤ, gQ (z : ZMod 10) = z • Q) :
    Function.Injective (gη.coprod gQ) := by
  classical
  rw [injective_iff_map_eq_zero]
  rintro ⟨a0, b0⟩ hker

  obtain ⟨a, rfl⟩ := ZMod.intCast_surjective a0
  obtain ⟨b, rfl⟩ := ZMod.intCast_surjective b0

  have hker' : (a • η + b • Q : H) = 0 := by
    simpa [AddMonoidHom.coprod_apply, hgη a, hgQ b] using hker

  by_cases ha0 : (a : ZMod 2) = 0
  · have haη : a • η = 0 := by
      have h := congrArg gη ha0
      simpa [hgη a] using h

    have hbQ : b • Q = 0 := by
      simpa [haη] using hker'

    have hbdiv' : ((addOrderOf Q : ℕ) : ℤ) ∣ b := by
      rw [addOrderOf_dvd_iff_zsmul_eq_zero]
      exact hbQ

    have hbdiv : (10 : ℤ) ∣ b := by
      simpa [hQ] using hbdiv'

    have hb0 : (b : ZMod 10) = 0 := by
      exact (ZMod.intCast_zmod_eq_zero_iff_dvd b 10).mpr hbdiv

    exact Prod.ext ha0 hb0

  · have ha1 : (a : ZMod 2) = 1 := by
      have hcases : (a : ZMod 2) = 0 ∨ (a : ZMod 2) = 1 := by
        let za : ZMod 2 := a
        have hza : (a : ZMod 2) = za := rfl
        fin_cases za
        · exact Or.inl hza
        · exact Or.inr hza
      exact hcases.resolve_left ha0

    have haη : a • η = η := by
      have h1 : gη (1 : ZMod 2) = η := by
        simpa using hgη (1 : ℤ)
      have h := congrArg gη ha1
      simpa [hgη a, h1] using h

    have hrel : η + b • Q = 0 := by
      simpa [haη] using hker'

    have hmem : η ∈ AddSubgroup.zmultiples Q := by
      rw [AddSubgroup.mem_zmultiples_iff]
      refine ⟨-b, ?_⟩
      simpa [neg_zsmul] using (eq_neg_of_add_eq_zero_left hrel).symm

    exact (hindep (eq_five_nsmul_of_order_two_in_zmultiples hη hQ hmem)).elim


theorem no_rational_point_of_order_20
    (E : WeierstrassCurve ℚ) [E.IsElliptic] :
    ¬ HasRationalPointOfOrder E 20 := by
  classical
  rintro ⟨P, hP⟩

  have h10P : addOrderOf ((10 : ℕ) • P) = 2 := by
    rw [addOrderOf_nsmul' (n := 10) P
      (by norm_num : (10 : ℕ) ≠ 0), hP]
    norm_num

  obtain ⟨E', hE', phi, eta, hη, hker, hdual⟩ :=
    exists_rational_two_isogeny_quotient ((10 : ℕ) • P) h10P
  letI : E'.IsElliptic := hE'

  let D : RationalTwoIsogenyData ((10 : ℕ) • P) :=
    ⟨E', hE', phi, eta, hη, hker, hdual⟩

  let Q := D.phi P

  have hηD : addOrderOf D.eta = 2 := by
    simpa [D] using hη

  have hQ : addOrderOf Q = 10 := by
    change addOrderOf (D.phi P) = 10
    exact image_order_10_of_order_20 hP D

  have hindep : D.eta ≠ 5 • Q := by
    change D.eta ≠ 5 • D.phi P
    exact eta_ne_half_image_20 hP D

  have hηzero : (2 : ℕ) • D.eta = 0 := by
    simpa [hηD] using addOrderOf_nsmul_eq_zero D.eta

  have hQzero : (10 : ℕ) • Q = 0 := by
    simpa [hQ] using addOrderOf_nsmul_eq_zero Q

  let gη : ZMod 2 →+ _ :=
    ZMod.lift 2 ⟨intSmulHom D.eta, by
      change ((2 : ℤ) • D.eta) = 0
      rwa [Nat.cast_smul_eq_nsmul ℤ]
    ⟩

  let gQ : ZMod 10 →+ _ :=
    ZMod.lift 10 ⟨intSmulHom Q, by
      change ((10 : ℤ) • Q) = 0
      rwa [Nat.cast_smul_eq_nsmul ℤ]
    ⟩

  have hgη : ∀ z : ℤ, gη (z : ZMod 2) = z • D.eta := by
    intro z
    simp [gη, intSmulHom]

  have hgQ : ∀ z : ℤ, gQ (z : ZMod 10) = z • Q := by
    intro z
    simp [gQ, intSmulHom]

  let f : ZMod 2 × ZMod 10 →+ _ := gη.coprod gQ

  have hf : Function.Injective f := by
    change Function.Injective (gη.coprod gQ)
    exact zmod_two_ten_coprod_injective hηD hQ hindep gη gQ hgη hgQ

  exact (no_Z2_cross_Z10_from_descent E') ⟨f, hf⟩
```
