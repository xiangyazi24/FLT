# Q3572: raw injectivity for current thetaRep

This targets the current representative convention in `[0, 2*T)`:

```lean
0 -> 0
root -> T
0 <= y -> 2*T - sigma x
y < 0 -> sigma x
```

The proof should be split into bucket lemmas. That keeps the final injectivity theorem small and avoids relying on a missing `StrictAntiOn.injectiveOn` API.

## 1. Sigma injectivity on `Ioi e`

```lean
import FLT.Assumptions.MazurProof.RealTopologyS5

noncomputable section
open Set
open scoped Topology

namespace MazurProof.RealTopology

theorem sigma_injective_of_right
    {A B e x x' : ℝ}
    (hroot : shortCubic A B e = 0)
    (hderiv : 0 < shortCubicDeriv A B e)
    (hposRight : ∀ {u : ℝ}, e < u -> 0 < shortCubic A B u)
    (hx : e < x) (hx' : e < x')
    (hsigma : sigma A B x = sigma A B x') :
    x = x' := by
  by_contra hne
  rcases lt_or_gt_of_ne hne with hlt | hgt
  · have hanti := strictAntiOn_sigma_Ioi
      (A := A) (B := B) (e := e) hroot hderiv hposRight
    have hs : sigma A B x' < sigma A B x := hanti hx hx' hlt
    linarith
  · have hanti := strictAntiOn_sigma_Ioi
      (A := A) (B := B) (e := e) hroot hderiv hposRight
    have hs : sigma A B x < sigma A B x' := hanti hx' hx hgt
    linarith

end MazurProof.RealTopology
```

If the call to `hanti` does not elaborate, replace `hanti hx hx' hlt` by:

```lean
hanti (by simpa : x ∈ Ioi e) (by simpa : x' ∈ Ioi e) hlt
```

## 2. Sign-refined branch lemmas

```lean
namespace MazurProof.RealTopology

theorem componentKer_some_y_eq_sqrt_of_nonneg
    {A B e x y : ℝ} {hpt}
    {hroot : shortCubic A B e = 0}
    {hderiv : 0 < shortCubicDeriv A B e}
    (hposRight : ∀ {u : ℝ}, e < u -> 0 < shortCubic A B u)
    {P : ComponentKer (A := A) (B := B) (e := e) hroot hderiv}
    (hP : P.1 = Point.some x y hpt)
    (hex : e < x) (hy : 0 <= y) :
    y = Real.sqrt (shortCubic A B x) := by
  rcases componentKer_some_y_branch_of_gt
      (A := A) (B := B) (e := e) (P := P) hP hex with h | h
  · exact h
  · have hsqrt_pos : 0 < Real.sqrt (shortCubic A B x) :=
      Real.sqrt_pos.2 (hposRight hex)
    have hyneg : y < 0 := by
      rw [h]
      nlinarith
    linarith

theorem componentKer_some_y_eq_neg_sqrt_of_neg
    {A B e x y : ℝ} {hpt}
    {hroot : shortCubic A B e = 0}
    {hderiv : 0 < shortCubicDeriv A B e}
    (hposRight : ∀ {u : ℝ}, e < u -> 0 < shortCubic A B u)
    {P : ComponentKer (A := A) (B := B) (e := e) hroot hderiv}
    (hP : P.1 = Point.some x y hpt)
    (hex : e < x) (hy : y < 0) :
    y = -Real.sqrt (shortCubic A B x) := by
  rcases componentKer_some_y_branch_of_gt
      (A := A) (B := B) (e := e) (P := P) hP hex with h | h
  · have hsqrt_pos : 0 < Real.sqrt (shortCubic A B x) :=
      Real.sqrt_pos.2 (hposRight hex)
    have hypos : 0 < y := by
      rw [h]
      exact hsqrt_pos
    linarith
  · exact h

theorem point_some_ext
    {A B x y x' y' : ℝ} {h h'}
    (hx : x = x') (hy : y = y') :
    (Point.some x y h : Point (shortW A B)) = Point.some x' y' h' := by
  rw [WeierstrassCurve.Affine.Point.some.injEq]
  exact ⟨hx, hy⟩

end MazurProof.RealTopology
```

## 3. Bucket range lemmas

```lean

theorem thetaRep_some_gt_lower_bounds
    {A B e x y : ℝ} {hpt}
    {hroot : shortCubic A B e = 0}
```

## 4. Zero bucket and final theorem shape

The zero bucket is the key helper for the outer `cases` proof.

```lean
namespace MazurProof.RealTopology

@[simp] theorem thetaRep_eq_zero_of_point_eq_zero
    {A B e : ℝ}
    {hroot : shortCubic A B e = 0}
    {hderiv : 0 < shortCubicDeriv A B e}
    {P : ComponentKer (A := A) (B := B) (e := e) hroot hderiv}
    (hP : P.1 = 0) : thetaRep P = 0 := by
  simp [thetaRep, hP]

theorem thetaRep_eq_zero_iff_point_eq_zero
    {A B e : ℝ}
    {hroot : shortCubic A B e = 0}
    {hderiv : 0 < shortCubicDeriv A B e}
    (hposRight : ∀ {u : ℝ}, e < u -> 0 < shortCubic A B u)
    (P : ComponentKer (A := A) (B := B) (e := e) hroot hderiv) :
    thetaRep P = 0 ↔ P.1 = 0 := by
  constructor
  · intro htheta
    cases hP : P.1 with
    | zero => simpa [hP]
    | some x y hpt =>
        rcases componentKer_some_eq_or_gt
            (A := A) (B := B) (e := e) (P := P) hP with hxroot | hex
        · have hrep := thetaRep_some_root
            (A := A) (B := B) (e := e) (P := P) hP hxroot
          have hT := halfPeriod_pos
            (A := A) (B := B) (e := e) hroot hderiv hposRight
          rw [hrep] at htheta
          nlinarith
        · by_cases hy : 0 <= y
          · have hb := thetaRep_some_gt_upper_bounds
              (A := A) (B := B) (e := e) hposRight (P := P) hP hex hy
            rw [htheta] at hb
            have hT := halfPeriod_pos
              (A := A) (B := B) (e := e) hroot hderiv hposRight
            nlinarith
          · have hb := thetaRep_some_gt_lower_bounds
              (A := A) (B := B) (e := e) hposRight (P := P) hP hex (lt_of_not_ge hy)
            rw [htheta] at hb
            nlinarith
  · intro hP
    exact thetaRep_eq_zero_of_point_eq_zero hP

end MazurProof.RealTopology
```

For the affine/affine core, use the following four cases after `componentKer_some_eq_or_gt` on both points and `by_cases hy : 0 <= y`, `by_cases hy' : 0 <= y'`:

```lean
-- upper / upper
have hsigma : sigma A B x = sigma A B x' := by
  rw [thetaRep_some_upper (P := P) hP hex hy,
      thetaRep_some_upper (P := Q) hQ hex' hy'] at htheta
  nlinarith
have hxx' := sigma_injective_of_right hroot hderiv hposRight hex hex' hsigma
have hy_s := componentKer_some_y_eq_sqrt_of_nonneg hposRight (P := P) hP hex hy
have hy'_s := componentKer_some_y_eq_sqrt_of_nonneg hposRight (P := Q) hQ hex' hy'
exact point_some_ext hxx' (by rw [hy_s, hy'_s, hxx'])

-- lower / lower
have hylt : y < 0 := lt_of_not_ge hy
have hylt' : y' < 0 := lt_of_not_ge hy'
have hsigma : sigma A B x = sigma A B x' := by
  rw [thetaRep_some_lower (P := P) hP hex hylt,
      thetaRep_some_lower (P := Q) hQ hex' hylt'] at htheta
  exact htheta
have hxx' := sigma_injective_of_right hroot hderiv hposRight hex hex' hsigma
have hy_s := componentKer_some_y_eq_neg_sqrt_of_neg hposRight (P := P) hP hex hylt
have hy'_s := componentKer_some_y_eq_neg_sqrt_of_neg hposRight (P := Q) hQ hex' hylt'
exact point_some_ext hxx' (by rw [hy_s, hy'_s, hxx'])

-- upper / lower contradiction
have hbP := thetaRep_some_gt_upper_bounds hposRight (P := P) hP hex hy
have hbQ := thetaRep_some_gt_lower_bounds hposRight (P := Q) hQ hex' (lt_of_not_ge hy')
rw [htheta] at hbP
nlinarith

-- lower / upper contradiction
have hbP := thetaRep_some_gt_lower_bounds hposRight (P := P) hP hex (lt_of_not_ge hy)
have hbQ := thetaRep_some_gt_upper_bounds hposRight (P := Q) hQ hex' hy'
rw [htheta] at hbP
nlinarith
```

Root/root uses `componentKer_some_y_eq_zero_of_x_eq_root` on both sides and `point_some_ext`. Root/right and right/root contradict the bucket bounds exactly as above: root has value `T`; lower has value `< T`; upper has value `> T`.

Then the final theorems are:

```lean
namespace MazurProof.RealTopology

theorem thetaRep_injective
    {A B e : ℝ}
    {hroot : shortCubic A B e = 0}
    {hderiv : 0 < shortCubicDeriv A B e}
    (hposRight : ∀ {u : ℝ}, e < u -> 0 < shortCubic A B u) :
    Function.Injective
      (thetaRep (A := A) (B := B) (e := e)
        (hroot := hroot) (hderiv := hderiv)) := by
  intro P Q htheta
  apply Subtype.ext
  cases hP : P.1 with
  | zero =>
      have hQtheta : thetaRep Q = 0 := by
        rw [<- htheta]
        exact thetaRep_eq_zero_of_point_eq_zero hP
      have hQzero :=
        (thetaRep_eq_zero_iff_point_eq_zero
          (A := A) (B := B) (e := e) hposRight Q).mp hQtheta
      simp [hP, hQzero]
  | some x y hpt =>
      cases hQ : Q.1 with
      | zero =>
          have hPtheta : thetaRep P = 0 := by
            rw [htheta]
            exact thetaRep_eq_zero_of_point_eq_zero hQ
          have hPzero :=
            (thetaRep_eq_zero_iff_point_eq_zero
              (A := A) (B := B) (e := e) hposRight P).mp hPtheta
          simp [hPzero, hQ]
      | some x' y' hpt' =>
          -- Use the affine/affine core described above.
          -- Recommended local lemma:
          --   thetaRep_same_affine_point hposRight hP hQ htheta
          -- returning `(Point.some x y hpt : Point (shortW A B)) = Point.some x' y' hpt'`.
          have hsome := thetaRep_same_affine_point
            (A := A) (B := B) (e := e) hposRight
            (P := P) (Q := Q) hP hQ htheta
          simpa [hP, hQ] using hsome

theorem thetaCandidate_injective
    {A B e : ℝ}
    {hroot : shortCubic A B e = 0}
    {hderiv : 0 < shortCubicDeriv A B e}
    (hposRight : ∀ {u : ℝ}, e < u -> 0 < shortCubic A B u) :
    Function.Injective
      (thetaCandidate (A := A) (B := B) (e := e)
        (hroot := hroot) (hderiv := hderiv)) := by
  intro P Q hPQ
  apply thetaRep_injective
      (A := A) (B := B) (e := e)
      (hroot := hroot) (hderiv := hderiv) hposRight
  exact (thetaCandidate_eq_iff_thetaRep_eq
      (A := A) (B := B) (e := e)
      (hroot := hroot) (hderiv := hderiv)
      hposRight P Q).mp hPQ

end MazurProof.RealTopology
```
