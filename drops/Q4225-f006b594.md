# N15 auxiliary curve by a quartic infinite descent

The file cited in the question, `scratch/FermatFourthDifferenceN16.lean`, and the declaration `no_coprime_fourth_difference` are not present on the connector-visible `scratch` branch.  The proof below therefore does not import either one.  It uses a direct minimal-hypotenuse descent on

```text
r^2 = m^4 + 62 m^2 n^2 + n^4
    = (m^2-n^2)^2 + (8mn)^2.
```

The key point is that **both** parity branches return a strictly smaller solution of the same quartic:

* if `m,n` are odd, divide the displayed Pythagorean triple by `8` and parameterize it;
* if `m,n` have opposite parity, parameterize the primitive triple, split the coprime rectangle `U*O=m*n`, and obtain two nested primitive Pythagorean triples.  The common square factor divides `15` and is `1 mod 8`, hence is `1`.

The code is self-contained modulo `Mathlib`, uses neither Mordell--Weil finite generation nor a new axiom, and contains no `sorry`.

```lean
import Mathlib

set_option maxHeartbeats 0

noncomputable section

namespace MazurProof.N15Auxiliary

/-- The auxiliary elliptic curve used in the order-15 exclusion. -/
def N15AuxiliaryEquation (X Y : ℚ) : Prop :=
  Y ^ 2 = X * (X - 15) * (X - 16)

/-- The seven affine rational points on the auxiliary curve. -/
def N15AuxiliaryAffineCandidate (X Y : ℚ) : Prop :=
  (X = 0 ∧ Y = 0) ∨
  (X = 15 ∧ Y = 0) ∨
  (X = 16 ∧ Y = 0) ∨
  (X = 12 ∧ Y = 12) ∨
  (X = 12 ∧ Y = -12) ∨
  (X = 20 ∧ Y = 20) ∨
  (X = 20 ∧ Y = -20)

/-! ## Pure algebra: the dual quartic -/

lemma n15_curve_to_dual_quartic
    (X Y : ℚ) (hX : X ≠ 0)
    (hcurve : N15AuxiliaryEquation X Y) :
    (240 / X - X) ^ 2 = (Y / X) ^ 4 + 62 * (Y / X) ^ 2 + 1 := by
  have hz : (Y / X) ^ 2 = X - 31 + 240 / X := by
    unfold N15AuxiliaryEquation at hcurve
    field_simp [hX]
    nlinarith [hcurve]
  calc
    (240 / X - X) ^ 2 = (X - 31 + 240 / X) ^ 2 +
        62 * (X - 31 + 240 / X) + 1 := by
      field_simp [hX]
      ring
    _ = (Y / X) ^ 4 + 62 * (Y / X) ^ 2 + 1 := by
      rw [← hz]
      ring

/-! ## Small arithmetic helpers -/

private lemma isCoprime_neg_left'
    {R : Type*} [CommRing R] {a b : R} (h : IsCoprime a b) :
    IsCoprime (-a) b := by
  rcases h with ⟨u, v, huv⟩
  refine ⟨-u, v, ?_⟩
  nlinarith

private lemma isCoprime_add_mul_left'
    {R : Type*} [CommRing R] {a b : R} (h : IsCoprime a b) (c : R) :
    IsCoprime (a + b * c) b := by
  rcases h with ⟨u, v, huv⟩
  refine ⟨u, v - u * c, ?_⟩
  calc
    u * (a + b * c) + (v - u * c) * b = u * a + v * b := by ring
    _ = 1 := huv

private lemma coprime_sq_sub_mul_int
    {m n : ℕ} (hcop : Nat.Coprime m n) :
    IsCoprime ((m : ℤ) ^ 2 - (n : ℤ) ^ 2) ((m : ℤ) * n) := by
  have hmn : IsCoprime (m : ℤ) (n : ℤ) := hcop.cast
  have hm0 : IsCoprime (-((n : ℤ) ^ 2)) (m : ℤ) :=
    isCoprime_neg_left' hmn.symm.pow_left
  have hm1 := isCoprime_add_mul_left' hm0 (m : ℤ)
  have hm : IsCoprime ((m : ℤ) ^ 2 - (n : ℤ) ^ 2) (m : ℤ) := by
    convert hm1 using 1 <;> ring
  have hn0 : IsCoprime ((m : ℤ) ^ 2) (n : ℤ) := hmn.pow_left
  have hn1 := isCoprime_add_mul_left' hn0 (-(n : ℤ))
  have hn : IsCoprime ((m : ℤ) ^ 2 - (n : ℤ) ^ 2) (n : ℤ) := by
    convert hn1 using 1 <;> ring
  simpa [mul_assoc] using hm.mul_right hn

private lemma eight_dvd_sq_sub_sq_of_odd
    {m n : ℤ} (hm : m % 2 = 1) (hn : n % 2 = 1) :
    (8 : ℤ) ∣ m ^ 2 - n ^ 2 := by
  have hm8 : m % 8 = 1 ∨ m % 8 = 3 ∨ m % 8 = 5 ∨ m % 8 = 7 := by
    omega
  have hn8 : n % 8 = 1 ∨ n % 8 = 3 ∨ n % 8 = 5 ∨ n % 8 = 7 := by
    omega
  apply Int.dvd_of_emod_eq_zero
  rcases hm8 with hm8 | hm8 | hm8 | hm8 <;>
    rcases hn8 with hn8 | hn8 | hn8 | hn8 <;>
    simp [pow_two, Int.sub_emod, Int.mul_emod, hm8, hn8]

private lemma nat_sq_mod_eight_of_odd
    {m : ℕ} (hm : m % 2 = 1) : m ^ 2 % 8 = 1 := by
  have hm8 : m % 8 = 1 ∨ m % 8 = 3 ∨ m % 8 = 5 ∨ m % 8 = 7 := by
    omega
  rcases hm8 with hm8 | hm8 | hm8 | hm8 <;>
    simp [pow_two, Nat.mul_mod, hm8]

private lemma factors_odd_of_product_odd
    {a b c : ℕ} (hc : c = a * b) (hodd : c % 2 = 1) :
    a % 2 = 1 ∧ b % 2 = 1 := by
  have hcne : ¬ Even c := (Nat.not_even_iff).2 hodd
  have habne : ¬ Even (a * b) := by simpa [hc] using hcne
  have ha : ¬ Even a := by
    intro h
    exact habne ((Nat.even_mul).2 (Or.inl h))
  have hb : ¬ Even b := by
    intro h
    exact habne ((Nat.even_mul).2 (Or.inr h))
  exact ⟨(Nat.not_even_iff).1 ha, (Nat.not_even_iff).1 hb⟩

private lemma coprime_sq_sum_self
    {a b : ℕ} (h : Nat.Coprime a b) :
    Nat.Coprime (a ^ 2 + b ^ 2) a := by
  rw [show a ^ 2 + b ^ 2 = a * a + b ^ 2 by ring,
    Nat.coprime_mul_left_add_left]
  exact h.symm.pow_left 2

private lemma common_factor_eq_one
    {h t : ℕ} (hh : 0 < h) (hcop : Nat.Coprime h t)
    (hdiv : h ∣ 15 * t ^ 2) (hmod : h % 8 = 1) : h = 1 := by
  have ht2 : Nat.Coprime h (t ^ 2) := hcop.pow_right 2
  have h15 : h ∣ 15 := (ht2.dvd_mul_right).1 hdiv
  have hle : h ≤ 15 := Nat.le_of_dvd (by decide) h15
  interval_cases h <;> norm_num at h15 hmod ⊢

private lemma four_dvd_left_of_mul_eq_four_mul
    {u v k : ℕ} (hv : v % 2 = 1) (h : u * v = 4 * k) : 4 ∣ u := by
  have hvOdd : Odd v := ⟨v / 2, by omega⟩
  have h2v : Nat.Coprime 2 v := Nat.coprime_two_left.2 hvOdd
  have h4v : Nat.Coprime 4 v := by
    convert h2v.pow_left 2 using 1 <;> norm_num
  apply (h4v.dvd_mul_right).1
  rw [h]
  exact dvd_mul_right 4 k

/-! A Nat-valued wrapper around Mathlib's primitive Pythagorean classification. -/
private lemma primitive_pythagorean_nat
    {x y z : ℕ} (hx : 0 < x) (hy : 0 < y) (hz : 0 < z)
    (hcop : Nat.Coprime x y) (hxodd : x % 2 = 1)
    (htrip : x ^ 2 + y ^ 2 = z ^ 2) :
    ∃ u v : ℕ,
      0 < v ∧ v < u ∧
      x + v ^ 2 = u ^ 2 ∧
      y = 2 * u * v ∧
      z = u ^ 2 + v ^ 2 ∧
      Nat.Coprime u v ∧
      ((u % 2 = 0 ∧ v % 2 = 1) ∨ (u % 2 = 1 ∧ v % 2 = 0)) := by
  have ht : PythagoreanTriple (x : ℤ) (y : ℤ) (z : ℤ) := by
    rw [PythagoreanTriple]
    exact_mod_cast (show x * x + y * y = z * z by simpa [pow_two] using htrip)
  have hcopZ : Int.gcd (x : ℤ) (y : ℤ) = 1 := by
    rw [Int.gcd_natCast_natCast]
    exact hcop.gcd_eq_one
  have hxoddZ : (x : ℤ) % 2 = 1 := by exact_mod_cast hxodd
  have hzZ : (0 : ℤ) < z := by exact_mod_cast hz
  obtain ⟨a, b, hxa, hyb, hzc, hab, hpar, ha0⟩ :=
    ht.coprime_classification' hcopZ hxoddZ hzZ
  have hxZ : (0 : ℤ) < x := by exact_mod_cast hx
  have hyZ : (0 : ℤ) < y := by exact_mod_cast hy
  have ha : 0 < a := by
    by_contra hna
    have : a = 0 := le_antisymm (le_of_not_gt hna) ha0
    subst a
    norm_num at hyb
    omega
  have hb : 0 < b := by nlinarith [hyb]
  have hba : b < a := by nlinarith [hxa]
  let u := a.toNat
  let v := b.toNat
  have hua : (u : ℤ) = a := by simp [u, Int.toNat_of_nonneg ha.le]
  have hvb : (v : ℤ) = b := by simp [v, Int.toNat_of_nonneg hb.le]
  have hu : 0 < u := Int.toNat_pos.2 ha
  have hv : 0 < v := Int.toNat_pos.2 hb
  have hvu : v < u := by exact_mod_cast hba
  have hxuvZ : (x : ℤ) + (v : ℤ) ^ 2 = (u : ℤ) ^ 2 := by
    rw [hua, hvb]
    nlinarith [hxa]
  have hxuv : x + v ^ 2 = u ^ 2 := by exact_mod_cast hxuvZ
  have hyv : y = 2 * u * v := by
    exact_mod_cast (show (y : ℤ) = 2 * (u : ℤ) * v by simpa [hua, hvb] using hyb)
  have hzuv : z = u ^ 2 + v ^ 2 := by
    exact_mod_cast (show (z : ℤ) = (u : ℤ) ^ 2 + v ^ 2 by simpa [hua, hvb] using hzc)
  have huv : Nat.Coprime u v := by
    rw [Nat.coprime_iff_gcd_eq_one, ← Int.gcd_natCast_natCast, hua, hvb]
    exact hab
  have hparNat :
      (u % 2 = 0 ∧ v % 2 = 1) ∨ (u % 2 = 1 ∧ v % 2 = 0) := by
    exact_mod_cast hpar
  exact ⟨u, v, hv, hvu, hxuv, hyv, hzuv, huv, hparNat⟩

/-! ## A coprime rectangle -/

private structure CoprimeRectangle (U O M N : ℕ) where
  a b c d : ℕ
  ha : 0 < a
  hb : 0 < b
  hc : 0 < c
  hd : 0 < d
  hU : U = a * d
  hO : O = b * c
  hM : M = a * b
  hN : N = c * d
  hab : Nat.Coprime a b
  hac : Nat.Coprime a c
  had : Nat.Coprime a d
  hbc : Nat.Coprime b c
  hbd : Nat.Coprime b d
  hcd : Nat.Coprime c d

private lemma coprime_rectangle
    {U O M N : ℕ}
    (hUpos : 0 < U) (hOpos : 0 < O) (hMpos : 0 < M) (hNpos : 0 < N)
    (hUO : Nat.Coprime U O) (hMN : Nat.Coprime M N)
    (hprod : U * O = M * N) : CoprimeRectangle U O M N := by
  let a := Nat.gcd U M
  let d := U / a
  let b := M / a
  have ha : 0 < a := Nat.gcd_pos_of_pos_left M hUpos
  have haU : a ∣ U := Nat.gcd_dvd_left U M
  have haM : a ∣ M := Nat.gcd_dvd_right U M
  have hUeq : a * d = U := by
    simpa [d] using Nat.mul_div_cancel' haU
  have hMeq : a * b = M := by
    simpa [b] using Nat.mul_div_cancel' haM
  have hd : 0 < d := by
    apply Nat.pos_of_ne_zero
    intro hd0
    simp [hd0] at hUeq
    omega
  have hb : 0 < b := by
    apply Nat.pos_of_ne_zero
    intro hb0
    simp [hb0] at hMeq
    omega
  have hdb : Nat.Coprime d b := by
    simpa [a, d, b] using Nat.coprime_div_gcd_div_gcd U M
  have hdo : d * O = b * N := by
    apply Nat.mul_left_cancel (n := a)
    calc
      a * (d * O) = (a * d) * O := by ring
      _ = U * O := by rw [hUeq]
      _ = M * N := hprod
      _ = (a * b) * N := by rw [hMeq]
      _ = a * (b * N) := by ring
  have hbO : b ∣ O := by
    apply (hdb.symm.dvd_mul_left).1
    rw [hdo]
    exact dvd_mul_right b N
  let c := O / b
  have hOeq : b * c = O := by
    simpa [c] using Nat.mul_div_cancel' hbO
  have hc : 0 < c := by
    apply Nat.pos_of_ne_zero
    intro hc0
    simp [hc0] at hOeq
    omega
  have hNeq : c * d = N := by
    apply Nat.mul_left_cancel (n := b)
    calc
      b * (c * d) = d * (b * c) := by ring
      _ = d * O := by rw [hOeq]
      _ = b * N := hdo
  have hdU : d ∣ U := ⟨a, by simpa [mul_comm] using hUeq.symm⟩
  have hbM : b ∣ M := ⟨a, by simpa [mul_comm] using hMeq.symm⟩
  have hcO : c ∣ O := ⟨b, by simpa [mul_comm] using hOeq.symm⟩
  have hcN : c ∣ N := ⟨d, hNeq.symm⟩
  have hdN : d ∣ N := ⟨c, by simpa [mul_comm] using hNeq.symm⟩
  have hab : Nat.Coprime a b := Nat.Coprime.of_dvd haU hbO hUO
  have hac : Nat.Coprime a c := Nat.Coprime.of_dvd haU hcO hUO
  have had : Nat.Coprime a d := Nat.Coprime.of_dvd haM hdN hMN
  have hbc : Nat.Coprime b c := Nat.Coprime.of_dvd hbM hcN hMN
  have hbd : Nat.Coprime b d := hdb.symm
  have hcd : Nat.Coprime c d := Nat.Coprime.of_dvd hcO hdU hUO.symm
  exact
    { a := a, b := b, c := c, d := d,
      ha := ha, hb := hb, hc := hc, hd := hd,
      hU := hUeq.symm, hO := hOeq.symm,
      hM := hMeq.symm, hN := hNeq.symm,
      hab := hab, hac := hac, had := had,
      hbc := hbc, hbd := hbd, hcd := hcd }

private lemma coprime_square_balance
    {b d F G : ℕ} (hb : 0 < b) (hd : 0 < d)
    (hbd : Nat.Coprime b d) (hFpos : 0 < F)
    (heq : b ^ 2 * F = d ^ 2 * G) :
    ∃ h : ℕ, 0 < h ∧ F = d ^ 2 * h ∧ G = b ^ 2 * h := by
  have hpow : Nat.Coprime (d ^ 2) (b ^ 2) := hbd.symm.pow 2 2
  have hdF : d ^ 2 ∣ F := by
    apply (hpow.dvd_mul_left).1
    rw [heq]
    exact dvd_mul_right (d ^ 2) G
  let h := F / d ^ 2
  have hF : d ^ 2 * h = F := by
    simpa [h] using Nat.mul_div_cancel' hdF
  have hG : G = b ^ 2 * h := by
    apply Nat.mul_left_cancel (n := d ^ 2)
    calc
      d ^ 2 * G = b ^ 2 * F := heq.symm
      _ = b ^ 2 * (d ^ 2 * h) := by rw [hF]
      _ = d ^ 2 * (b ^ 2 * h) := by ring
  have hh : 0 < h := by
    apply Nat.pos_of_ne_zero
    intro hh0
    simp [hh0] at hF
    omega
  exact ⟨h, hh, hF.symm, hG⟩

private def N15Bad (R : ℕ) : Prop :=
  ∃ M N : ℕ,
    0 < N ∧ N < M ∧ Nat.Coprime M N ∧
      R ^ 2 = M ^ 4 + 62 * M ^ 2 * N ^ 2 + N ^ 4

/-! ## Odd--odd descent -/

private lemma descend_odd_odd
    {R M N : ℕ}
    (hN : 0 < N) (hNM : N < M) (hMN : Nat.Coprime M N)
    (hM2 : M % 2 = 1) (hN2 : N % 2 = 1)
    (hQ : R ^ 2 = M ^ 4 + 62 * M ^ 2 * N ^ 2 + N ^ 4) :
    ∃ R' < R, N15Bad R' := by
  have hM : 0 < M := hN.trans hNM
  have hR : 0 < R := by nlinarith [show 0 < N ^ 4 by positivity]
  have hsqle : N ^ 2 ≤ M ^ 2 := Nat.pow_le_pow_left hNM.le 2
  have h8Z : (8 : ℤ) ∣ (M : ℤ) ^ 2 - (N : ℤ) ^ 2 :=
    eight_dvd_sq_sub_sq_of_odd (by exact_mod_cast hM2) (by exact_mod_cast hN2)
  have h8 : 8 ∣ M ^ 2 - N ^ 2 := by
    rw [← Int.natCast_dvd, Int.natCast_sub hsqle]
    exact h8Z
  let A := (M ^ 2 - N ^ 2) / 8
  let B := M * N
  have hA : 8 * A = M ^ 2 - N ^ 2 := by
    simpa [A] using Nat.mul_div_cancel' h8
  have hApos : 0 < A := by
    have : 0 < M ^ 2 - N ^ 2 := Nat.sub_pos_of_lt (Nat.pow_lt_pow_left hNM (by decide))
    omega
  have hBpos : 0 < B := by positivity
  have hDiffProd := coprime_sq_sub_mul_int hMN
  have hAdiv : (A : ℤ) ∣ (M : ℤ) ^ 2 - (N : ℤ) ^ 2 := by
    refine ⟨8, ?_⟩
    exact_mod_cast (show M ^ 2 - N ^ 2 = A * 8 by simpa [mul_comm] using hA.symm)
  have hABZ : IsCoprime (A : ℤ) (B : ℤ) := by
    simpa [B] using hDiffProd.of_isCoprime_of_dvd_left hAdiv
  have hAB : Nat.Coprime A B := hABZ.natCoprime
  have hscaled : (8 * A) ^ 2 + (8 * B) ^ 2 = R ^ 2 := by
    rw [hA]
    dsimp [B]
    nlinarith [hQ]
  have ht : PythagoreanTriple ((8 * A : ℕ) : ℤ) ((8 * B : ℕ) : ℤ) (R : ℤ) := by
    rw [PythagoreanTriple]
    exact_mod_cast (show (8 * A) * (8 * A) + (8 * B) * (8 * B) = R * R by
      simpa [pow_two] using hscaled)
  have h8g : 8 ∣ Int.gcd ((8 * A : ℕ) : ℤ) ((8 * B : ℕ) : ℤ) := by
    apply Nat.dvd_gcd <;> norm_num
  have h8RZ : (8 : ℤ) ∣ (R : ℤ) :=
    (Int.natCast_dvd.2 h8g).trans ht.gcd_dvd
  have h8R : 8 ∣ R := Int.natCast_dvd.1 h8RZ
  let C := R / 8
  have hC : 8 * C = R := by simpa [C] using Nat.mul_div_cancel' h8R
  have hCpos : 0 < C := by omega
  have htrip : A ^ 2 + B ^ 2 = C ^ 2 := by
    nlinarith [hscaled, hC]
  have hBodd : B % 2 = 1 := by
    simp [B, pow_two, Nat.mul_mod, hM2, hN2]
  obtain ⟨u, v, hv, hvu, hBuv, hAuv, hCuv, huv, hpar⟩ :=
    primitive_pythagorean_nat hBpos hApos hCpos hAB.symm hBodd
      (by nlinarith [htrip])
  have h2sum : 2 ∣ M ^ 2 + N ^ 2 := by
    apply Nat.dvd_of_mod_eq_zero
    simp [pow_two, Nat.add_mod, Nat.mul_mod, hM2, hN2]
  let S := (M ^ 2 + N ^ 2) / 2
  have hS : 2 * S = M ^ 2 + N ^ 2 := by
    simpa [S] using Nat.mul_div_cancel' h2sum
  have hdiff : M ^ 2 - N ^ 2 = 16 * u * v := by
    rw [← hA]
    rw [hAuv]
    ring
  have hprod : M * N = u ^ 2 - v ^ 2 := by
    dsimp [B] at hBuv
    omega
  have hdiffSq := congrArg (fun t : ℕ => t ^ 2) hdiff
  have hprodSq := congrArg (fun t : ℕ => t ^ 2) hprod
  have hSSq := congrArg (fun t : ℕ => t ^ 2) hS
  have hnew : S ^ 2 = u ^ 4 + 62 * u ^ 2 * v ^ 2 + v ^ 4 := by
    nlinarith [hdiffSq, hprodSq, hSSq]
  have hRformula : R ^ 2 = (M ^ 2 + N ^ 2) ^ 2 + 60 * M ^ 2 * N ^ 2 := by
    nlinarith [hQ]
  have hsumR : M ^ 2 + N ^ 2 < R := by
    by_contra hle
    have hsquares : R ^ 2 ≤ (M ^ 2 + N ^ 2) ^ 2 :=
      Nat.pow_le_pow_left (Nat.le_of_not_gt hle) 2
    nlinarith [hRformula, show 0 < M ^ 2 * N ^ 2 by positivity]
  have hSR : S < R := by omega
  exact ⟨S, hSR, u, v, hv, hvu, huv, hnew⟩

/-! ## Opposite-parity descent -/

private lemma descend_opposite
    {R M N : ℕ}
    (hN : 0 < N) (hNM : N < M) (hMN : Nat.Coprime M N)
    (hparMN : (M % 2 = 0 ∧ N % 2 = 1) ∨
      (M % 2 = 1 ∧ N % 2 = 0))
    (hQ : R ^ 2 = M ^ 4 + 62 * M ^ 2 * N ^ 2 + N ^ 4) :
    ∃ R' < R, N15Bad R' := by
  have hM : 0 < M := hN.trans hNM
  have hR : 0 < R := by nlinarith [show 0 < N ^ 4 by positivity]
  have hsqle : N ^ 2 ≤ M ^ 2 := Nat.pow_le_pow_left hNM.le 2
  let A := M ^ 2 - N ^ 2
  let B := 8 * M * N
  have hArel : A + N ^ 2 = M ^ 2 := by
    dsimp [A]
    exact Nat.sub_add_cancel hsqle
  have hApos : 0 < A := by
    dsimp [A]
    exact Nat.sub_pos_of_lt (Nat.pow_lt_pow_left hNM (by decide))
  have hBpos : 0 < B := by positivity
  have hMmod : M ^ 2 % 2 = M % 2 := by
    rcases hparMN with h | h <;> simp [pow_two, Nat.mul_mod, h.1]
  have hNmod : N ^ 2 % 2 = N % 2 := by
    rcases hparMN with h | h <;> simp [pow_two, Nat.mul_mod, h.2]
  have hAodd : A % 2 = 1 := by omega
  have hAodd' : Odd A := ⟨A / 2, by omega⟩
  have hA8 : Nat.Coprime A 8 := by
    have hA2 : Nat.Coprime A 2 := Nat.coprime_two_right.2 hAodd'
    convert hA2.pow_right 3 using 1 <;> norm_num
  have hDiffProd := coprime_sq_sub_mul_int hMN
  have hAcast : (A : ℤ) = (M : ℤ) ^ 2 - (N : ℤ) ^ 2 := by
    dsimp [A]
    rw [Int.natCast_sub hsqle]
  have hAProdZ : IsCoprime (A : ℤ) ((M * N : ℕ) : ℤ) := by
    rw [hAcast]
    simpa using hDiffProd
  have hAProd : Nat.Coprime A (M * N) := hAProdZ.natCoprime
  have hAB : Nat.Coprime A B := by
    simpa [B, mul_assoc] using hA8.mul_right hAProd
  have htrip : A ^ 2 + B ^ 2 = R ^ 2 := by
    dsimp [A, B]
    nlinarith [hQ]

private theorem no_N15Bad : ¬ ∃ R : ℕ, N15Bad R := by
  rintro ⟨R0, hR0⟩
  let S : Set ℕ := {R | N15Bad R}
  have hS : S.Nonempty := ⟨R0, hR0⟩
  let R := Nat.find hS
  have hR : N15Bad R := Nat.find_spec hS
  rcases hR with ⟨M, N, hN, hNM, hMN, hQ⟩
  rcases Nat.mod_two_eq_zero_or_one M with hM2 | hM2 <;>
    rcases Nat.mod_two_eq_zero_or_one N with hN2 | hN2
  · exfalso
    apply Nat.not_coprime_of_dvd_of_dvd (by decide : 1 < 2)
      (Nat.dvd_of_mod_eq_zero hM2) (Nat.dvd_of_mod_eq_zero hN2) hMN
  · obtain ⟨R', hR'R, hbad⟩ :=
      descend_opposite hN hNM hMN (Or.inl ⟨hM2, hN2⟩) hQ
    have hmin : R ≤ R' := Nat.find_min' hS hbad

  · obtain ⟨R', hR'R, hbad⟩ :=

    have hmin : R ≤ R' := Nat.find_min' hS hbad

theorem n15_dual_quartic_exhaustion_int

    change Nat.gcd M N = 1
    simpa [M, N, Int.gcd] using hcop

  have hN2 : (N : ℤ) ^ 2 = n ^ 2 := by simp [N, pow_two]
  have hR2 : (R : ℤ) ^ 2 = r ^ 2 := by simp [R, pow_two]
  have hNat : R ^ 2 = M ^ 4 + 62 * M ^ 2 * N ^ 2 + N ^ 4 := by
    apply Int.ofNat_inj.1
    push_cast
    nlinarith [h, hM2, hN2, hR2]
```
