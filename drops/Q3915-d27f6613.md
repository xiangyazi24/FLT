# Q3915 — Injecting `ZMod 2 × ZMod 10` from two independent torsion points

Use `ZMod.lift` for each cyclic factor, then use `AddMonoidHom.coprod` to add the two images.  `AddMonoidHom.prodMap` is the wrong direction for the final map: it maps a product to a product.  Here the desired map is the coproduct

```lean
(a, b) ↦ f₂ a + f₁₀ b.
```

The proof of injectivity is best organized through the kernel.  The only nontrivial group-theoretic point is the helper lemma `eq_five_nsmul_of_order_two_mem_zmultiples`: in the cyclic subgroup generated by an element of additive order `10`, the unique nonzero element of order dividing `2` is `5 • Q`.

```lean
import Mathlib

noncomputable section

open Function

variable {H : Type*} [AddCommGroup H]

/-- The additive hom `ℤ →+ H`, `z ↦ z • x`. -/
def zsmulFromIntHom (x : H) : ℤ →+ H where
  toFun z := z • x
  map_zero' := by simp
  map_add' a b := by
    simpa using (add_zsmul x a b)

/-- The hom `ZMod n →+ H` sending `1` to `x`, given a proof `n • x = 0`. -/
def zmodCyclicHom (n : ℕ) (x : H) (hx : n • x = 0) : ZMod n →+ H :=
  ZMod.lift n ⟨zsmulFromIntHom x, by
    change ((n : ℤ) • x) = 0
    rwa [Nat.cast_smul_eq_nsmul ℤ]⟩

@[simp]
lemma zmodCyclicHom_intCast (n : ℕ) (x : H) (hx : n • x = 0) (z : ℤ) :
    zmodCyclicHom n x hx (z : ZMod n) = z • x := by
  simp [zmodCyclicHom, zsmulFromIntHom]

@[simp]
lemma zmodCyclicHom_zero (n : ℕ) (x : H) (hx : n • x = 0) :
    zmodCyclicHom n x hx 0 = 0 := by
  simp

@[simp]
lemma zmodCyclicHom_one (n : ℕ) (x : H) (hx : n • x = 0) :
    zmodCyclicHom n x hx 1 = x := by
  simpa using zmodCyclicHom_intCast n x hx (1 : ℤ)

/-- In a cyclic additive subgroup generated by an element of order `10`, the only element of
additive order `2` is `5 • Q`. -/
theorem eq_five_nsmul_of_order_two_mem_zmultiples
    {η Q : H}
    (hη : addOrderOf η = 2)
    (hQ : addOrderOf Q = 10)
    (hmem : η ∈ AddSubgroup.zmultiples Q) :
    η = 5 • Q := by
  classical
  have hQfin : IsOfFinAddOrder Q := by
    rw [← addOrderOf_pos_iff]
    rw [hQ]
    norm_num
  have hmem_range :
      η ∈ (Finset.range (addOrderOf Q)).image (fun d : ℕ => d • Q) :=
    (hQfin.mem_zmultiples_iff_mem_range_addOrderOf).mp hmem
  rw [hQ] at hmem_range
  rcases Finset.mem_image.mp hmem_range with ⟨d, hd_range, hdη⟩
  have hdlt : d < 10 := by
    simpa using Finset.mem_range.mp hd_range
  have hηeq : η = d • Q := hdη.symm

  have h2η : (2 : ℕ) • η = 0 := by
    simpa [hη] using addOrderOf_nsmul_eq_zero η
  have h2dQ : (2 * d) • Q = 0 := by
    simpa [hηeq, nsmul_nsmul] using h2η
  have hdvd : 10 ∣ 2 * d := by
    rw [← hQ, addOrderOf_dvd_iff_nsmul_eq_zero]
    exact h2dQ
  have hd05 : d = 0 ∨ d = 5 := by
    rcases hdvd with ⟨k, hk⟩
    omega
  have hη_ne_zero : η ≠ 0 := by
    intro h0
    have hbad := hη
    rw [h0] at hbad
    norm_num at hbad
  have hd_ne_zero : d ≠ 0 := by
    intro hd0
    apply hη_ne_zero
    simpa [hηeq, hd0]
  have hd5 : d = 5 := by
    rcases hd05 with hd0 | hd5
    · exact (hd_ne_zero hd0).elim
    · exact hd5
  simpa [hηeq, hd5]

/-- The desired homomorphism `ZMod 2 × ZMod 10 →+ H`, sending `(1,0)` to `η` and `(0,1)` to `Q`. -/
def torsionProductHom
    (η Q : H)
    (hη : addOrderOf η = 2)
    (hQ : addOrderOf Q = 10) :
    ZMod 2 × ZMod 10 →+ H :=
  let hη₂ : (2 : ℕ) • η = 0 := by
    simpa [hη] using addOrderOf_nsmul_eq_zero η
  let hQ₁₀ : (10 : ℕ) • Q = 0 := by
    simpa [hQ] using addOrderOf_nsmul_eq_zero Q
  (zmodCyclicHom 2 η hη₂).coprod (zmodCyclicHom 10 Q hQ₁₀)

@[simp]
theorem torsionProductHom_apply
    (η Q : H)
    (hη : addOrderOf η = 2)
    (hQ : addOrderOf Q = 10)
    (x : ZMod 2 × ZMod 10) :
    torsionProductHom η Q hη hQ x =
      zmodCyclicHom 2 η (by simpa [hη] using addOrderOf_nsmul_eq_zero η) x.1
        + zmodCyclicHom 10 Q (by simpa [hQ] using addOrderOf_nsmul_eq_zero Q) x.2 := by
  cases x
  rfl

@[simp]
theorem torsionProductHom_apply_one_zero
    (η Q : H)
    (hη : addOrderOf η = 2)
    (hQ : addOrderOf Q = 10) :
    torsionProductHom η Q hη hQ (1, 0) = η := by
  simp [torsionProductHom]

@[simp]
theorem torsionProductHom_apply_zero_one
    (η Q : H)
    (hη : addOrderOf η = 2)
    (hQ : addOrderOf Q = 10) :
    torsionProductHom η Q hη hQ (0, 1) = Q := by
  simp [torsionProductHom]

/-- The homomorphism is injective if `η` is not the order-2 element `5 • Q` in `⟨Q⟩`. -/
theorem torsionProductHom_injective
    {η Q : H}
    (hη : addOrderOf η = 2)
    (hQ : addOrderOf Q = 10)
    (hη_ne_fiveQ : η ≠ 5 • Q) :
    Function.Injective (torsionProductHom η Q hη hQ) := by
  classical
  let hη₂ : (2 : ℕ) • η = 0 := by
    simpa [hη] using addOrderOf_nsmul_eq_zero η
  let hQ₁₀ : (10 : ℕ) • Q = 0 := by
    simpa [hQ] using addOrderOf_nsmul_eq_zero Q
  rw [injective_iff_map_eq_zero]
  rintro ⟨a0, b0⟩ hker
  obtain ⟨a, rfl⟩ := ZMod.intCast_surjective a0
  obtain ⟨b, rfl⟩ := ZMod.intCast_surjective b0
  have hker' : (a • η + b • Q : H) = 0 := by
    simpa [torsionProductHom, zmodCyclicHom, zsmulFromIntHom] using hker

  by_cases ha0 : (a : ZMod 2) = 0
  · have haη : a • η = 0 := by
      have hc := congrArg (zmodCyclicHom 2 η hη₂) ha0
      simpa [zmodCyclicHom, zsmulFromIntHom] using hc
    have hbQ : b • Q = 0 := by
      simpa [haη] using hker'
    have hb0 : (b : ZMod 10) = 0 := by
      rw [ZMod.intCast_zmod_eq_zero_iff_dvd]
      change ((10 : ℕ) : ℤ) ∣ b
      rw [← hQ, addOrderOf_dvd_iff_zsmul_eq_zero]
      exact hbQ
    simpa [ha0, hb0]
  · have ha1 : (a : ZMod 2) = 1 := by
      have hcases : (a : ZMod 2) = 0 ∨ (a : ZMod 2) = 1 := by
        fin_cases (a : ZMod 2) <;> simp
      exact hcases.resolve_left ha0
    have haη : a • η = η := by
      have hc := congrArg (zmodCyclicHom 2 η hη₂) ha1
      simpa [zmodCyclicHom, zsmulFromIntHom] using hc
    have hrel : η + b • Q = 0 := by
      simpa [haη] using hker'
    have hmem : η ∈ AddSubgroup.zmultiples Q := by
      rw [AddSubgroup.mem_zmultiples_iff]
      refine ⟨-b, ?_⟩
      calc
        η = -(b • Q) := eq_neg_of_add_eq_zero_left hrel
        _ = (-b) • Q := by simp
    exact (hη_ne_fiveQ (eq_five_nsmul_of_order_two_mem_zmultiples hη hQ hmem)).elim

/-- A packaged version: the constructed hom together with the two generator computations and
injectivity. -/
theorem exists_injective_torsionProductHom
    {η Q : H}
    (hη : addOrderOf η = 2)
    (hQ : addOrderOf Q = 10)
    (hη_ne_fiveQ : η ≠ 5 • Q) :
    ∃ f : ZMod 2 × ZMod 10 →+ H,
      Function.Injective f ∧ f (1, 0) = η ∧ f (0, 1) = Q := by
  refine ⟨torsionProductHom η Q hη hQ, ?_, ?_, ?_⟩
  · exact torsionProductHom_injective hη hQ hη_ne_fiveQ
  · simp
  · simp
```

## Lemmas used

The construction relies on these Mathlib APIs.

```lean
#check ZMod.lift
#check ZMod.lift_coe
#check AddMonoidHom.coprod
#check addOrderOf_nsmul_eq_zero
#check addOrderOf_dvd_iff_nsmul_eq_zero
#check addOrderOf_dvd_iff_zsmul_eq_zero
#check IsOfFinAddOrder.mem_zmultiples_iff_mem_range_addOrderOf
#check AddSubgroup.mem_zmultiples_iff
#check ZMod.intCast_surjective
#check ZMod.intCast_zmod_eq_zero_iff_dvd
```

The main reason this is shorter than a direct proof with `ZMod.val` is that `ZMod.intCast_surjective` lets the kernel element be represented as `((a : ZMod 2), (b : ZMod 10))`; then `ZMod.lift_coe` immediately evaluates the map as `a • η + b • Q`.
